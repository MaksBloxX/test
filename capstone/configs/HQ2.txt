! v4.0 build-along for HQ2 (not executed in Packet Tracer)
! Cable exactly as in capstone/plan.json; adjust interface labels if your PT module differs.
! For serial clock rate, the HQ cable end MUST be DCE (show controllers serial).
enable
configure terminal
hostname HQ2
no ip domain-lookup
interface GigabitEthernet0/0
 description 802.1Q trunk to HQ-SW GigabitEthernet0/2
 no ip address
 no shutdown
exit
interface GigabitEthernet0/0.10
 description HQ-DATA HSRP group 10
 encapsulation dot1Q 10
 ip address 10.10.10.3 255.255.255.0
 standby 10 ip 10.10.10.1
 standby 10 priority 100
 standby 10 preempt
 ip nat inside
 no shutdown
exit
interface GigabitEthernet0/0.20
 description HQ-VOICE HSRP group 20
 encapsulation dot1Q 20
 ip address 10.10.20.3 255.255.255.0
 standby 20 ip 10.10.20.1
 standby 20 priority 100
 standby 20 preempt
 ip nat inside
 no shutdown
exit
interface GigabitEthernet0/0.40
 description HQ-DMZ HSRP group 40
 encapsulation dot1Q 40
 ip address 10.10.40.3 255.255.255.0
 standby 40 ip 10.10.40.1
 standby 40 priority 100
 standby 40 preempt
 ip nat inside
 no shutdown
exit
interface GigabitEthernet0/0.99
 description HQ-ADMIN HSRP group 99
 encapsulation dot1Q 99
 ip address 10.10.99.3 255.255.255.0
 standby 99 ip 10.10.99.1
 standby 99 priority 100
 standby 99 preempt
 ip nat inside
 no shutdown
exit
interface Serial0/0/0
 description Private point-to-point to BR1 Serial0/0/1
 ip address 10.255.12.1 255.255.255.252
 clock rate 64000
 ip nat inside
 no shutdown
exit
interface Serial0/0/1
 description Private point-to-point to BR2 Serial0/0/1
 ip address 10.255.22.1 255.255.255.252
 clock rate 64000
 ip nat inside
 no shutdown
exit
interface GigabitEthernet0/1
 description ISP test link to GigabitEthernet0/1
 ip address 198.51.100.5 255.255.255.252
 ip nat outside
 no shutdown
exit
ip route 0.0.0.0 0.0.0.0 198.51.100.6
router ospf 10
 router-id 2.2.2.2
 passive-interface default
 no passive-interface Serial0/0/0
 no passive-interface Serial0/0/1
 network 10.10.10.0 0.0.0.255 area 0
 network 10.10.20.0 0.0.0.255 area 0
 network 10.10.40.0 0.0.0.255 area 0
 network 10.10.99.0 0.0.0.255 area 0
 network 10.255.12.0 0.0.0.3 area 0
 network 10.255.22.0 0.0.0.3 area 0
exit
! NAT overload is only for traffic leaving the ISP interface; ISP does not run private OSPF.
access-list 1 permit 10.0.0.0 0.255.255.255
ip nat inside source list 1 interface GigabitEthernet0/1 overload
! Only HTTP is intentionally published. DNS, mail and FTP are private DMZ services.
ip nat inside source static tcp 10.10.40.20 80 198.51.100.5 80
end
copy running-config startup-config
